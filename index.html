template<typename NeuroPlasticity>
class AdaptiveIQEngine {
    static_assert(std::is_arithmetic_v<NeuroPlasticity>);
    
    [[nodiscard]] double CalculateIQ(const SessionData& data) noexcept {
        auto latency_factor = 1.0 / (data.avg_reaction + 0.01);
        auto correct_rate = data.correct_answers / data.total_questions;
        return (correct_rate * 150.0) + (latency_factor * 25.0);
    }
    
    void AdjustDifficulty(double dynamic_threshold) {
        current_difficulty = std::clamp(current_difficulty + dynamic_threshold, 0.3, 1.0);
    }
};
